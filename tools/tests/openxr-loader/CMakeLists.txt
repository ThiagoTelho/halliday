cmake_minimum_required(VERSION 3.25)
project(HallidayOfficialLoaderSmoke LANGUAGES C CXX)

if(NOT WIN32)
  message(FATAL_ERROR "Use an x86_64 Windows cross compiler for this smoke test")
endif()
get_filename_component(HALLIDAY_REPO "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(HALLIDAY_OPENXR_SDK "${HALLIDAY_REPO}/reference/openxr-sdk" CACHE PATH "Official OpenXR SDK sources")
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_API_LAYERS OFF CACHE BOOL "" FORCE)
set(BUILD_CONFORMANCE_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_WITH_SYSTEM_JSONCPP OFF CACHE BOOL "" FORCE)
set(DYNAMIC_LOADER ON CACHE BOOL "" FORCE)
set(CMAKE_DISABLE_FIND_PACKAGE_OpenGL ON CACHE BOOL "" FORCE)
set(CMAKE_DISABLE_FIND_PACKAGE_Vulkan ON CACHE BOOL "" FORCE)
add_subdirectory("${HALLIDAY_OPENXR_SDK}" sdk)

# SDK 1.1.65 applies its ELF version script to MinGW. Use its Windows export list
# instead. These target properties leave the official loader sources unchanged.
if(MINGW)
  set_target_properties(openxr_loader PROPERTIES LINK_FLAGS "" PREFIX "")
  target_sources(openxr_loader PRIVATE "${HALLIDAY_OPENXR_SDK}/src/loader/openxr-loader.def")
  target_link_options(openxr_loader PRIVATE -static)
endif()
set_target_properties(openxr_loader PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

add_executable(openxr-loader-smoke "${CMAKE_CURRENT_LIST_DIR}/../openxr-loader-smoke.cpp")
target_compile_features(openxr-loader-smoke PRIVATE cxx_std_17)
target_compile_definitions(openxr-loader-smoke PRIVATE XR_USE_GRAPHICS_API_D3D11)
target_include_directories(openxr-loader-smoke PRIVATE "${HALLIDAY_REPO}/windows-openxr/src")
target_link_libraries(openxr-loader-smoke PRIVATE openxr_loader d3d11 dxgi dxguid)
target_link_options(openxr-loader-smoke PRIVATE -static -municode)
set_target_properties(openxr-loader-smoke PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
