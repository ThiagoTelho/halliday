cmake_minimum_required(VERSION 3.25)
project(HallidayOpenXRRouter LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_path(HALLIDAY_JSON_INCLUDE nlohmann/json.hpp REQUIRED)
add_executable(router_config_tests tests/config_tests.cpp)
target_include_directories(router_config_tests PRIVATE . "${HALLIDAY_JSON_INCLUDE}")
enable_testing()
add_test(NAME router_config COMMAND router_config_tests)
if(WIN32)
  add_library(halliday_openvr_router SHARED openvr_router.cpp)
  target_include_directories(halliday_openvr_router PRIVATE "${HALLIDAY_JSON_INCLUDE}")
  target_compile_options(halliday_openvr_router PRIVATE -Wall -Wextra -Werror)
  target_link_options(halliday_openvr_router PRIVATE -static)
  set_target_properties(halliday_openvr_router PROPERTIES PREFIX "" OUTPUT_NAME "vrclient_x64")
  add_library(halliday_openxr_router SHARED router.cpp)
  target_include_directories(halliday_openxr_router PRIVATE ../include "${HALLIDAY_JSON_INCLUDE}")
  target_compile_options(halliday_openxr_router PRIVATE -Wall -Wextra -Werror)
  target_link_options(halliday_openxr_router PRIVATE -static)
  set_target_properties(halliday_openxr_router PROPERTIES PREFIX "")
  file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/halliday_openxr_router.json" CONTENT
"{\n  \"file_format_version\": \"1.0.0\",\n  \"runtime\": {\"library_path\":\"./halliday_openxr_router.dll\"}\n}\n")
endif()
