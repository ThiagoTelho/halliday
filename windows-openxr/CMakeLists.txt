cmake_minimum_required(VERSION 3.25)
project(HallidayWindowsOpenXR VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(HALLIDAY_XR_TIMING_DIAGNOSTIC_DEFAULT "Enable frame timing output in an explicit diagnostic build" OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

if(WIN32)
  add_library(halliday_openxr SHARED
    src/entrypoints.cpp src/runtime.cpp src/system.cpp src/graphics.cpp
    src/swapchain.cpp src/frame.cpp src/space.cpp src/input.cpp src/path.cpp
    src/control.cpp src/frame_timing.cpp src/d3d11_readback.cpp src/d3d12_readback.cpp src/panel_readback.cpp src/panel_readback_d3d12.cpp src/vulkan_graphics.cpp src/vulkan_transfer.cpp src/halliday_openxr.def)

  target_include_directories(halliday_openxr PRIVATE include src "${HALLIDAY_VULKAN_INCLUDE_DIR}")
  target_compile_definitions(halliday_openxr PRIVATE XR_NO_PROTOTYPES XR_USE_GRAPHICS_API_D3D11 XR_USE_GRAPHICS_API_D3D12 XR_USE_GRAPHICS_API_VULKAN)
  target_compile_options(halliday_openxr PRIVATE -Wall -Wextra -Werror=return-type)
  if(HALLIDAY_XR_TIMING_DIAGNOSTIC_DEFAULT)
    target_compile_definitions(halliday_openxr PRIVATE HALLIDAY_XR_TIMING_DIAGNOSTIC_DEFAULT=1)
  endif()
  target_link_libraries(halliday_openxr PRIVATE d3d11 d3d12 dxgi dxguid)
  target_link_options(halliday_openxr PRIVATE -static)
  set_target_properties(halliday_openxr PROPERTIES PREFIX "")

  target_link_options(control_input_concurrency_test PRIVATE -static)
  add_executable(d3d11_prior_staging_test tests/d3d11_prior_staging_test.cpp src/d3d11_readback.cpp src/frame_timing.cpp)
  target_include_directories(d3d11_prior_staging_test PRIVATE include src)
  target_link_options(d3d11_prior_staging_test PRIVATE -static)
  add_executable(d3d12_prior_staging_test tests/d3d12_prior_staging_test.cpp src/d3d12_readback.cpp src/d3d11_readback.cpp src/frame_timing.cpp)
  target_include_directories(d3d12_prior_staging_test PRIVATE include src)
  target_link_libraries(d3d12_prior_staging_test PRIVATE dxguid)
  target_link_options(d3d12_prior_staging_test PRIVATE -static)
  file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/bin/halliday_openxr.json" CONTENT
"{\n  \"file_format_version\": \"1.0.0\",\n  \"runtime\": {\n    \"library_path\": \"./halliday_openxr.dll\",\n    \"functions\": {\"xrNegotiateLoaderRuntimeInterface\": \"xrNegotiateLoaderRuntimeInterface\"}\n  }\n}\n")
  add_executable(runtime_vulkan_smoke tests/runtime_vulkan_smoke.cpp)

  target_include_directories(d3d12_fence_failure PRIVATE include src)
  target_compile_definitions(d3d12_fence_failure PRIVATE XR_NO_PROTOTYPES XR_USE_GRAPHICS_API_D3D11 XR_USE_GRAPHICS_API_D3D12)
  target_link_libraries(d3d12_fence_failure PRIVATE d3d12 dxgi dxguid)
  target_link_options(d3d12_fence_failure PRIVATE -static)

  target_include_directories(runtime_smoke PRIVATE include src)
  target_compile_definitions(runtime_smoke PRIVATE XR_NO_PROTOTYPES XR_USE_GRAPHICS_API_D3D11)
  target_link_libraries(runtime_smoke PRIVATE d3d11 dxgi dxguid)
  target_link_options(runtime_smoke PRIVATE -static -municode)
  add_executable(runtime_idle_mapping tests/runtime_idle_mapping.cpp)
  target_include_directories(runtime_idle_mapping PRIVATE include src)
  target_compile_definitions(runtime_idle_mapping PRIVATE XR_NO_PROTOTYPES XR_USE_GRAPHICS_API_D3D11)
  target_link_libraries(runtime_idle_mapping PRIVATE d3d11 dxgi dxguid)

    target_link_options(runtime_d3d12_smoke PRIVATE -static -municode)
